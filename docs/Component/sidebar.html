Supported Renderers in v13

TYPE: "SIDEBAR" Sidebar Navigation

Mounts responsive Bootstrap-styled sidebar navigation into target element. Supports items like Dashboard, Orders, and Products with automated icons and active link states.

render({ type: "sidebar", data, targetHtmlId: "sidebarContainer" })
Second Image Content · sidebar.js ES Module 
import render from "../dist/v13/min.js";
import data from "./sidebar.json" with { type: "json" };

const start = () => {
  try {
    render({
      type: "sidebar",
      data,
      targetHtmlId: "sidebarContainer"
    });
  } catch (err) {
    console.log("error : ", err);
  }
};

start();
