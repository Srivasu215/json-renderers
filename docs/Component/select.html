TYPE: "SELECT" Select Element

Mounts a newly created <select id="LedgerName"> populated with child option elements into the target container.

render({ type: "select", targetHtmlId, data })
Second Image Content · select.js ES Module 
import render from "../dist/v13/min.js";
import data from "./select.json" with { type: "json" };

const start = () => {
  try {
    render({
      type: "select",
      data,
      targetHtmlId: "selectContainer"
    });
  } catch (err) {
    console.log("error : ", err);
  }
};

start();
