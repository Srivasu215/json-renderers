Supported Renderers in v13

TYPE: "SELECTOPTIONSONLY" Option Injection

Populates dynamic option items directly into a pre-existing <select> element without modifying outer wrapper tags.

render({ type: "selectOptionsOnly", targetHtmlId, data })
Second Image Content · selectOptionsOnly.js ES Module 
import render from "../dist/v13/min.js";
import data from "./selectOptionsOnly.json" with { type: "json" };

const start = () => {
  try {
    render({
      type: "selectOptionsOnly",
      data,
      targetHtmlId: "selectId"
    });
  } catch (err) {
    console.log("error : ", err);
  }
};

start();
