TYPE: "TABLE" Data Grid

Mounts complete Bootstrap-styled table into the target element. Supports multi-column headers and automated cell value bindings.

render({ type: "table", targetHtmlId, data, columns })
Second Image Content · table.js ES Module 
import render from "../dist/v13/min.js";
import data from "./table.json" with { type: "json" };

const start = () => {
  try {
    render({
      type: "table",
      data,
      targetHtmlId: "tableContainer",
      columns: ["id", "name", "category", "status"]
    });
  } catch (err) {
    console.log("error : ", err);
  }
};

start();
